# Host build of pmor_core + offline render + tests (no libDaisy, no ARM toolchain). Firmware: ./Makefile.
cmake_minimum_required(VERSION 3.16)
project(pmor_host C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Bit-exact across CPUs/compilers' fused-multiply-add habits: no contraction, no fast-math.
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
add_compile_options(-ffp-contract=off -fno-fast-math)

# shared tap analyzer + 14-bit tempo wire format (one copy, shared with the Pi transport; see Makefile)

file(GLOB CORE_SRC "${CMAKE_SOURCE_DIR}/pmor_core/*.cpp")
add_library(pmor_core STATIC ${CORE_SRC} pmor_core/host/platform_host.cpp)
target_include_directories(pmor_core PUBLIC pmor_core ${CMAKE_SOURCE_DIR} ${CMAKE_SOURCE_DIR}/pmor_core/third_party)
target_compile_definitions(pmor_core PUBLIC PMOR_HOST=1)

add_executable(pmor_render tests/render_main.cpp tests/scenario.cpp)
target_link_libraries(pmor_render pmor_core)
target_include_directories(pmor_render PRIVATE tests)

add_executable(f32stat tests/f32stat.cpp)
add_executable(test_schema tests/test_schema.cpp)
target_link_libraries(test_schema pmor_core)
add_executable(test_patch_roundtrip tests/test_patch_roundtrip.cpp)
target_link_libraries(test_patch_roundtrip pmor_core)
target_compile_definitions(test_patch_roundtrip PRIVATE PATCH_EXPECT_SIZE=7540)   # v8 = frozen v7 (7476, measured on ff86e4c) + 16 floats drum_eng

add_executable(pmorctl tools/pmorctl.cpp)
target_link_libraries(pmorctl pmor_core)
